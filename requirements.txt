# 1. MEDIA, VIDEO & AUDIO PROCESSING (System CLI & Python)
# =============================================================================
# FFmpeg (System CLI - Executed via shell/subprocess):
#  - Purpose: Complete video and audio manipulation, transcoding, clipping, merging, and subtitle burning.
#  - Trim / Cut video (fast, lossless): ffmpeg -ss 00:01:00 -to 00:02:30 -i input.mp4 -c copy output.mp4
#  - Burn hard subtitles (SRT/ASS) into video: ffmpeg -i input.mp4 -vf "subtitles=subs.srt:force_style='FontName=DejaVu Sans,FontSize=24'" output.mp4
#  - Extract audio from video (for STT/analysis): ffmpeg -i input.mp4 -vn -acodec libmp3lame -q:a 2 audio.mp3
#  - Resize / Compress video: ffmpeg -i input.mp4 -vf scale=1280:-2 -crf 23 -preset fast output.mp4
#  - Merge video + audio: ffmpeg -i video.mp4 -i audio.mp3 -c:v copy -c:a aac output.mp4
#
# yt-dlp (Python / CLI):
#  - Purpose: Search, inspect and download videos, audio, thumbnails, metadata and subtitles from YouTube and 1000+ sites.
#    This is THE tool for anything on YouTube (research a topic, get a transcript, fetch a clip for ffmpeg).
#  - Search:     yt-dlp "ytsearch5:QUERY" --flat-playlist --print "%(title)s | %(webpage_url)s | %(duration_string)s"
#  - Info:       yt-dlp -J URL            (JSON: title, description, chapters, duration, formats)
#  - Transcript: yt-dlp --write-auto-subs --sub-langs "tr,en" --skip-download -o "%(id)s" URL
#  - Download:   yt-dlp -f "bv*[height<=720]+ba/b" -o "%(title)s.%(ext)s" URL      (audio only: -x --audio-format mp3)
#  - Python:     yt_dlp.YoutubeDL(opts).extract_info(url, download=False)
yt-dlp>=2024.0.0
#
# edge-tts (Python):
#  - Purpose: Free, high-quality neural Text-to-Speech (TTS) using Microsoft Edge's online voices (supports Turkish tr-TR-AhmetNeural / tr-TR-EmelNeural).
#  - Usage: Generate .mp3 voiceovers for videos, notifications, or spoken agent responses.
edge-tts>=7.0
#
# faster-whisper (Python):
#  - Purpose: offline speech-to-text on the CPU (int8) for the speech_to_text and inspect_media tools: transcripts
